{

4. Enter the file name from the user and the data from the user and copy that file into another file.

Code
#include <stdio.h>

void main()
{
FILE *fp1,*fp2;
char file1[50],file2[50],ch;

printf("Enter source file: ");
scanf("%s",file1);

printf("Enter destination file: ");
scanf("%s",file2);

fp1=fopen(file1,"r");
fp2=fopen(file2,"w");

while((ch=fgetc(fp1))!=EOF)
{
fputc(ch,fp2);
}

fclose(fp1);
fclose(fp2);

printf("File copied successfully");
}
Output
Enter source file: a.txt
Enter destination file: b.txt
File copied successfully

7. Create a file and count no. of characters in a file.

Code

void main()
{
FILE *fp;

8. Create a file and count no. of words in a file.

Code
#include <stdio.h>
#include <ctype.h>

void main()
{
FILE *fp;
char fname[50],ch;
int words=0,inword=0;

printf("Enter file name: ");
scanf("%s",fname);

fp=fopen(fname,"r");

while((ch=fgetc(fp))!=EOF)
{
if(!isspace(ch) && inword==0)
{
words++;
inword=1;
}
else if(isspace(ch))
{
inword=0;
}
}

fclose(fp);

printf("Total Words=%d",words);
}
Output
Enter file name: test.txt
Hello World From C Programming
Total Words=5
